claude mcp add --transport http jt-doc-tools {{ url }} --header "Authorization: Bearer <API_TOKEN>"

  

  

    
{{ tr('寫進 `~/.codex/config.toml`，並設定環境變數 `JTDT_TOKEN`：')|marks }}

    


[mcp_servers.jt-doc-tools]
url = "{{ url }}"
bearer_token_env_var = "JTDT_TOKEN"

    


export JTDT_TOKEN=<API_TOKEN>

  

  

    
{{ tr('寫進 `opencode.json`，並設定環境變數 `JTDT_TOKEN`：')|marks }}

    


{
  "$schema": "https://opencode.ai/config.json",
  "mcp": {
    "jt-doc-tools": {
      "type": "remote",
      "url": "{{ url }}",
      "enabled": true,
      "oauth": false,
      "headers": { "Authorization": "Bearer {env:JTDT_TOKEN}" }
    }
  }
}

  

  

    
{{ tr('寫進 `claude_desktop_config.json`（透過 mcp-remote 轉接，需要 Node.js）：')|marks }}

    


{
  "mcpServers": {
    "jt-doc-tools": {
      "command": "npx",
      "args": ["-y", "mcp-remote", "{{ url }}", "--header", "Authorization:${JTDT_AUTH}"],
      "env": { "JTDT_AUTH": "Bearer <API_TOKEN>" }
    }
  }
}
